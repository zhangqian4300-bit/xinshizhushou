// services/chat.service.js
export class ChatService {
  constructor(db, memoryService) {
    this.db = db;
    this.memoryService = memoryService;
  }

  /**
   * 处理对话消息
   * @param {string} userId 用户ID
   * @param {string} message 用户消息
   * @returns {Object} AI响应 { message, suggestions, carePlan? }
   */
  async handleChat(userId, message) {
    // 1. 获取用户画像
    const profile = await this.memoryService.getUserProfile(userId);

    // 2. 获取最近对话历史
    const history = await this.memoryService.getRecentHistory(userId, 10);

    // 3. 构建Prompt（注入记忆）
    const prompt = this.buildPrompt(profile, history, message);

    // 4. 调用OpenClaw Gateway
    const aiResponse = await this.invokeAI(userId, prompt);

    // 5. 解析AI响应
    const parsed = this.parseAIResponse(aiResponse);

    // 6. 保存对话记录
    await this.memoryService.saveConversation(userId, message, parsed);

    // 7. 如果有养护计划更新，保存
    if (parsed.carePlan) {
      await this.memoryService.saveCarePlan(userId, parsed.carePlan);
    }

    return parsed;
  }

  /**
   * 构建带记忆的Prompt
   */
  buildPrompt(profile, history, message) {
    const historySummary = history
      .slice(-5)
      .map(h => `${h.role}: ${h.content}`)
      .join('\n');

    return `
【用户画像】
- 内耗类型：${profile.exhaustType}
- 测评时间：${profile.assessDate}

【最近对话】
${historySummary || '无'}

【当前消息】
${message}

请以心石清理师的身份回应，输出JSON格式。
    `.trim();
  }

  /**
   * 调用OpenClaw Gateway
   */
  async invokeAI(userId, prompt) {
    // 通过OpenClawBackend的_invoke方法调用
    // 这部分由框架处理
  }

  /**
   * 解析AI响应
   */
  parseAIResponse(aiResponse) {
    // AI输出格式：
    // {
    //   "message": "回应内容（Markdown）",
    //   "suggestions": ["建议1", "建议2"],
    //   "carePlan": { ... } // 可选
    // }
    try {
      return JSON.parse(aiResponse);
    } catch (e) {
      // 如果不是JSON，包装成默认格式
      return {
        message: aiResponse,
        suggestions: ['继续说说...', '我还有别的困惑']
      };

```javascript
// services/memory.service.js
export class MemoryService {
  constructor(db) {
    this.db = db;
  }

  /**
   * 获取用户画像
   */
  async getUserProfile(userId) {
    const sql = `SELECT * FROM user_profiles WHERE user_id = ?`;
    const row = await this.db.get(sql, [userId]);
    return row || null;
  }

  /**
   * 保存用户画像
   */
  async saveUserProfile(userId, profile) {
    const sql = `
      INSERT OR REPLACE INTO user_profiles
      (user_id, exhaust_type, assess_result, updated_at)
      VALUES (?, ?, ?, ?)
    `;
    await this.db.run(sql, [
      userId,
      profile.exhaustType,
      JSON.stringify(profile.assessResult),
      new Date().toISOString()
    ]);
  }

  /**
   * 获取最近对话历史
   */
  async getRecentHistory(userId, limit = 10) {
    const sql = `
      SELECT role, content, created_at
      FROM conversations
      WHERE user_id = ?
      ORDER BY created_at DESC
      LIMIT ?
    `;
    const rows = await this.db.all(sql, [userId, limit]);
    return rows.reverse(); // 按时间正序
  }

  /**
   * 保存对话记录
   */
  async saveConversation(userId, userMessage, aiResponse) {
    // 保存用户消息
    await this.db.run(`
      INSERT INTO conversations (user_id, role, content)
      VALUES (?, 'user', ?)
    `, [userId, userMessage]);

    // 保存AI响应
    await this.db.run(`
      INSERT INTO conversations (user_id, role, content)
      VALUES (?, 'assistant', ?)
    `, [userId, JSON.stringify(aiResponse)]);
  }

  /**
   * 保存养护计划
   */
  async saveCarePlan(userId, carePlan) {
    const sql = `
      INSERT OR REPLACE INTO care_plans
      (user_id, plan_data, updated_at)
      VALUES (?, ?, ?)
    `;
    await this.db.run(sql, [
      userId,
      JSON.stringify(carePlan),
      new Date().toISOString()
    ]);
  }

  /**
   * 获取养护计划
   */
  async getCarePlan(userId) {
    const sql = `SELECT plan_data FROM care_plans WHERE user_id = ?`;
    const row = await this.db.get(sql, [userId]);
    return row ? JSON.parse(row.plan_data) : null;
  }

  /**
   * 获取清理历史（旅程数据）
   */
  async getJourneyData(userId) {
    const sql = `
      SELECT
        DATE(created_at) as date,
        COUNT(CASE WHEN role = 'user' THEN 1 END) as messages,
        GROUP_CONCAT(
          CASE WHEN role = 'user' THEN content END, '|||'
        ) as user_messages
      FROM conversations
      WHERE user_id = ?
      GROUP BY DATE(created_at)
      ORDER BY date DESC
      LIMIT 30
    `;
    return await this.db.all(sql, [userId]);
  }
}
```

### 4.2 表结构定义

```sql
-- 用户画像表
CREATE TABLE user_profiles (
  user_id TEXT PRIMARY KEY,
  exhaust_type TEXT,                -- 内耗类型
  assess_result TEXT,               -- JSON: 测评答案和结果
  created_at TEXT DEFAULT (datetime('now')),
  updated_at TEXT DEFAULT (datetime('now'))
);

-- 对话记录表
CREATE TABLE conversations (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT NOT NULL,
  role TEXT NOT NULL,               -- 'user' or 'assistant'
  content TEXT NOT NULL,
  created_at TEXT DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES user_profiles(user_id)
);

-- 养护计划表
CREATE TABLE care_plans (
  user_id TEXT PRIMARY KEY,
  plan_data TEXT NOT NULL,          -- JSON: 计划内容
  updated_at TEXT DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES user_profiles(user_id)
);

-- 清理记录表（可选，用于旅程页面）
CREATE TABLE clear_records (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT NOT NULL,
  date TEXT NOT NULL,               -- 日期 YYYY-MM-DD
  stone_type TEXT,                  -- 心石类型
  content TEXT,                     -- 清理内容
  suggestion TEXT,                  -- AI建议
  created_at TEXT DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES user_profiles(user_id)
);

-- 索引
CREATE INDEX idx_conversations_user ON conversations(user_id);
CREATE INDEX idx_conversations_time ON conversations(created_at);
CREATE INDEX idx_clear_records_user ON clear_records(user_id);
```

## 六、AI Prompt设计

### 6.1 System Prompt

```markdown
# 角色定位

你是「心石清理师」AI助手，严格遵循北京协和医院张海敏博士提出的「减法养生」情绪自愈理念。

# 核心理念

最高级的养生不是往心里加东西，而是把心里的石头一块块搬开。

# 输出格式

你必须输出JSON格式：

```json
{
  "message": "你的回应内容，支持Markdown格式",
  "suggestions": ["建议用户说的话1", "建议2", "建议3"],
  "carePlan": {
    "tasks": [
      {
        "templateId": "模板ID",
        "customContent": {
          "action": "具体动作",
          "rationale": "为什么适合这个用户（引用对话）"
        }
      }
    ],
    "progressMessage": "进度提示语"
  }
}
```

# 养护计划模板

可选模板ID：
- morning_breath: 晨曦观息
- mindful_afternoon: 正念午后
- prune_relation: 修剪关系
- night_review: 深夜复盘
- language_watch: 语言觉察
- five_reduce: 五减默念

# 规则

1. 每次只给1-2个可落地的小动作，符合减法理念
2. rationale必须引用用户之前说过的话
3. suggestions简短，不超过15字
4. 不用"你应该"，直接引导
5. 可以使用Markdown格式：**加粗**、> 引用、- 列表

# 记忆运用

你会收到用户的画像和对话历史，请体现"我记得你"：
- 引用用户之前说过的话
- 关联用户的内耗类型
- 对比用户之前的状态变化
```

### 6.2 Prompt注入格式

```javascript
// 构建完整的Prompt
function buildFullPrompt(userProfile, history, message) {
  return `
${SYSTEM_PROMPT}

---

【用户画像】
- 内耗类型：${userProfile.exhaustType}
- 测评日期：${userProfile.assessDate}

【最近对话】
${formatHistory(history)}

【当前日期】
${new Date().toLocaleDateString('zh-CN')}

【用户消息】
${message}

请以JSON格式回应。
  `.trim();
}

function formatHistory(history) {
  if (!history || history.length === 0) return '无';

  return history.slice(-5).map(h => {
    if (h.role === 'user') {
      return `用户: ${h.content}`;
    }
    // AI消息可能包含JSON，提取message字段
    try {
      const parsed = JSON.parse(h.content);
      return `AI: ${parsed.message?.slice(0, 100) || h.content.slice(0, 100)}...`;
    } catch {
      return `AI: ${h.content.slice(0, 100)}...`;
    }
  }).join('\n');
}
```

## 八、安全设计

### 8.1 认证机制

```javascript
// 基于设备ID的用户识别
// 小程序：使用openid
// Web：生成UUID存储在localStorage

function getUserId() {
  let userId = localStorage.getItem('xinshi_user_id');
  if (!userId) {
    userId = 'web-' + crypto.randomUUID();
    localStorage.setItem('xinshi_user_id', userId);
  }
  return userId;
}
```

### 8.2 数据安全

- 用户数据按userId隔离
- 敏感内容不记录（如果涉及医疗诊断，引导去医院）
- 支持数据导出和删除（合规要求）

### 8.3 输入过滤

```javascript
// 简单的输入过滤
function sanitizeInput(input) {
  // 限制长度
  if (input.length > 2000) {
    input = input.slice(0, 2000);
  }
  // 移除危险字符（XSS防护）
  return input
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;');
}
